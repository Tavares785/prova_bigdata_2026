QUERY 2 - Top 5 clientes por gasto

SELECT c.cliente_nome, ROUND(SUM(f.valor_total), 2) AS gasto_total
FROM fato_pedidos f
JOIN dim_cliente c ON f.cliente_id = c.cliente_id
GROUP BY c.cliente_nome
ORDER BY gasto_total DESC
LIMIT 5;
